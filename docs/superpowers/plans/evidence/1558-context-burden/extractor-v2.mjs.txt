function walk(dir, pred, out = []) {
  if (!existsSync(dir)) return out;
  for (const ent of readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, ent.name);
    if (ent.isDirectory()) walk(p, pred, out);
    else if (pred(p)) out.push(p);
  }
  return out;
}

function relSource(p) {
  return p
    .replace('/Users/kpburson/.codex/', '~/.codex/')
    .replace('/Users/kpburson/.claude/', '~/.claude/');
}

function sha256(text) {
  return createHash('sha256').update(text).digest('hex');
}

function bytes(text) {
  return Buffer.byteLength(String(text ?? ''), 'utf8');
}

function textOf(v) {
  if (v == null) return '';
  if (typeof v === 'string') return v;
  if (Array.isArray(v)) return v.map(textOf).join('\n');
  if (typeof v === 'object') return Object.values(v).map(textOf).join('\n');
  return String(v);
}

function decodeJsonStringLiteral(s) {
  if (!s) return '';
  try {
    return JSON.parse(`"${s.replace(/"/g, '\\"')}"`);
  } catch {
    return s.replace(/\\n/g, '\n').replace(/\\"/g, '"');
  }
}

function beforeCutoff(timestamp) {
  if (!timestamp) return true;
  const ms = Date.parse(timestamp);
  return !Number.isFinite(ms) || ms <= EVENT_CUTOFF_MS;
}

function codexCommand(input) {
  const m = String(input ?? '').match(/"cmd"\s*:\s*"((?:\\.|[^"\\])*)"/);
  if (m) return decodeJsonStringLiteral(m[1]);
  return String(input ?? '');
}

function splitShell(s) {
  return String(s ?? '')
    .split(/\n|;/)
    .map((x) => x.trim())
    .filter(Boolean);
}

function instructionInfo(text) {
  for (const [kind, re] of instructionPatterns) {
    const m = String(text).match(re);
    if (m) return { kind, path: m[0] };
  }
  return null;
}

function lifecycleInfo(line) {
  const t = String(line).replace(/^cd\s+[^&]+&&\s*/, '').trim();
  if (/^\*\*\* Begin Patch|^apply_patch\b/.test(t)) return null;
  if (/\b--help\b|\bhelp\b/i.test(t)) return null;
  const md = t.match(moveDone);
  if (md) return { verb: 'close', issue: md[1], commandFamily: 'move-state-done' };
  if (createIssue.test(t)) return { verb: 'create-issue', issue: null, commandFamily: 'create-issue' };
  const m = t.match(lifecycleStart);
  if (!m) return null;
  const first = m[1].replace(/^#/, '');
  if (/^\d+$/.test(first) && new RegExp(`\\baitm\\s+#?${first}\\s*>`).test(t)) return null;
  const rest = String(m[2] ?? '')
    .trim()
    .split(/\s+/)
    .filter((x) => !/^\d?>|^&?\d$|^2>&1$/.test(x))
    .map((x) => x.replace(/^#/, ''));
  const verb = /^\d+$/.test(first) ? 'bind' : first;
  const issue = /^\d+$/.test(first) ? first : rest.find((x) => /^\d+$/.test(x));
  return { verb, issue: issue ?? null, commandFamily: 'aitm' };
}

function classify(line, toolName = '') {
  const text = `${line} ${toolName}`;
  const tool = String(toolName ?? '');
  const isPatch = /apply_patch/i.test(tool);
  const isReadLike = /\b(Read|read_file|open|exec|Bash|Shell|shell|command)\b/i.test(tool) || !tool;
  if (isPatch) {
    return { category: 'other_unattributable', subcategory: 'other-visible-tool-traffic', instruction: instructionInfo(text), lifecycle: null };
  }
  const inst = instructionInfo(text);
  const life = isReadLike ? lifecycleInfo(line) : null;
  if (life) return { category: 'lifecycle_command', subcategory: life.verb, instruction: null, lifecycle: life };
  if (inst && isReadLike && /^(?:sed|cat|head|tail|less|Read\b)|file_path|SKILL\.md|router\.md|pickup-directive|session-boot/.test(text)) {
    return { category: 'instruction_loading', subcategory: inst.kind, instruction: inst, lifecycle: null };
  }
  if (isReadLike && (explicitHelp.test(text) || ordinarySearch.test(text)) && workflowTerms.test(text)) {
    return { category: 'help_discovery', subcategory: explicitHelp.test(text) ? 'command-help' : 'workflow-guidance-search', instruction: inst, lifecycle: null };
  }
  return { category: 'other_unattributable', subcategory: 'other-visible-tool-traffic', instruction: inst, lifecycle: null };
}

function outputCompleteness(text) {
  const s = String(text ?? '');
  if (!s) return 'unavailable';
  if (/Warning: truncated output|truncated output|tokens truncated|omitted/i.test(s)) return 'truncated';
  return 'complete';
}

function closeOutcome(command, output) {
  const life = lifecycleInfo(command);
  if (!life || life.verb !== 'close') return 'not_close';
  const s = String(output ?? '').replace(ansi, '');
  if (!s) return 'attempted_close_unknown_output';
  if (/Script failed|Exit code [1-9]|error:|refus(?:e|ed|al)|blocked|unchecked|not approved|failed/i.test(s)) return 'failed_close';
  if (/Done|closed|moved .*Done|Story closed|Final Review Passed|Timing data flushed/i.test(s)) return 'confirmed_successful_close';
  return 'attempted_close_unknown_outcome';
}

function commandOutcome(command, output) {
  const close = closeOutcome(command, output);
  if (close !== 'not_close') return close;
  const s = String(output ?? '');
  if (!s) return 'unknown_output';
  return 'output_present';
}

function visibleHookText(att) {
  const chunks = [];
  if (att?.content) chunks.push(textOf(att.content));
  if (att?.stdout) chunks.push(String(att.stdout));
  if (att?.stderr) chunks.push(String(att.stderr));
  if (att?.rendered) chunks.push(textOf(att.rendered));
  return chunks.join('\n');
}

const events = [];

function addEvent(e) {
  e.inputChars = e.inputChars ?? [...String(e.inputText ?? '')].length;
  e.inputBytes = e.inputBytes ?? bytes(e.inputText ?? '');
  e.outputChars = e.outputChars ?? [...String(e.outputText ?? '')].length;
  e.outputBytes = e.outputBytes ?? bytes(e.outputText ?? '');
  e.proxyInputTokens = Math.ceil(e.inputChars / 4);
  e.proxyOutputTokens = Math.ceil(e.outputChars / 4);
  e.inputHash = sha256(String(e.inputText ?? ''));
  e.outputHash = sha256(String(e.outputText ?? ''));
  e.contentHash = sha256(`${e.provider}\n${e.sessionId}\n${e.category}\n${e.inputHash}\n${e.outputHash}`);
  delete e.inputText;
  delete e.outputText;
  events.push(e);
}

function processCodex(file) {
  const text = readFileSync(file, 'utf8');
  const lines = text.split(/\n/);
  let sessionId = null;
  let cwd = '';
  let relevant = false;
  for (const line of lines) {
    if (!line) continue;
    try {
      const o = JSON.parse(line);
      if (!beforeCutoff(o.timestamp)) continue;
      if (o.type === 'session_meta') {
        sessionId = o.payload?.session_id ?? o.payload?.id ?? sessionId;
        cwd = o.payload?.cwd ?? cwd;
        if (cwd.includes('ai-task-manager')) relevant = true;
      }
    } catch {}
  }
  if (!relevant) return;
  const calls = new Map();
  let segment = 0;
  let activeIssue = null;
  const segmentInstructionSeen = new Set();
  const issues = new Set();
  manifest.push({ provider: 'codex', sourceLog: relSource(file), sessionId, cwd, bytes: bytes(text), sha256: sha256(text), lines: lines.length });
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    if (!line) continue;
    let o;
    try { o = JSON.parse(line); } catch { continue; }
    if (!beforeCutoff(o.timestamp)) continue;
    const loc = { sourceLog: relSource(file), line: i + 1, ordinal: o.ordinal ?? null };
    if (o.type === 'compacted') {
      segment += 1;
      activeIssue = null;
      segmentInstructionSeen.clear();
      addEvent({
        provider: 'codex', sessionId, timestamp: o.timestamp ?? null, contextSegment: segment,
        issue: null, attributionMethod: 'unattributed', category: 'compaction_reset',
        subcategory: 'codex-compacted-record', tool: 'compacted', command: null,
        instructionPath: null, instructionLoadOrdinalInSegment: null, outcome: 'segment_boundary',
        completeness: 'complete', duplicateStatus: 'unchecked', ...loc, inputText: '', outputText: ''
      });
      continue;
    }
    if (o.type === 'session_meta' && o.payload?.base_instructions?.text) {
      addEvent({
        provider: 'codex', sessionId, timestamp: o.timestamp ?? o.payload?.timestamp ?? null, contextSegment: segment,
        issue: null, attributionMethod: 'unattributed', category: 'host_hook_injected',
        subcategory: 'base-instructions', tool: 'session_meta', command: null,
        instructionPath: null, instructionLoadOrdinalInSegment: null, outcome: 'visible_in_log',
        completeness: 'complete', duplicateStatus: 'unchecked', ...loc, inputText: o.payload.base_instructions.text, outputText: ''
      });
    }
    if (o.type === 'world_state' && o.payload?.state?.agents_md?.text) {
      addEvent({
        provider: 'codex', sessionId, timestamp: o.timestamp ?? null, contextSegment: segment,
        issue: activeIssue, attributionMethod: activeIssue ? 'active-binding' : 'unattributed',
        category: 'host_hook_injected', subcategory: 'AGENTS.md-world-state', tool: 'world_state',
        command: null, instructionPath: 'AGENTS.md', instructionLoadOrdinalInSegment: null,
        outcome: 'visible_in_log', completeness: 'complete', duplicateStatus: 'unchecked', ...loc,
        inputText: o.payload.state.agents_md.text, outputText: ''
      });
    }
    if (o.type === 'response_item' && o.payload?.type === 'message' && o.payload.role === 'developer') {
      const t = textOf(o.payload.content);
      if (/<skills_instructions>|<app-context>|<permissions instructions>|<multi_agent_mode>/.test(t)) {
        addEvent({
          provider: 'codex', sessionId, timestamp: o.timestamp ?? null, contextSegment: segment,
          issue: null, attributionMethod: 'unattributed', category: 'host_hook_injected',
          subcategory: 'developer-instructions', tool: 'developer-message', command: null,
          instructionPath: null, instructionLoadOrdinalInSegment: null, outcome: 'visible_in_log',
          completeness: outputCompleteness(t), duplicateStatus: 'unchecked', ...loc, inputText: t, outputText: ''
        });
      }
    }
    if (o.type === 'response_item' && o.payload?.type === 'custom_tool_call') {
      const command = codexCommand(o.payload.input);
      const parts = splitShell(command);
      const primary = parts.find((p) => classify(p, o.payload.name).category !== 'other_unattributable') ?? parts[0] ?? command;
      const c = classify(primary, o.payload.name);
      const life = c.lifecycle;
      const issue = life?.issue ?? activeIssue;
      if (life?.issue) {
        issues.add(life.issue);
        if (['bind', 'start', 'resume'].includes(life.verb)) activeIssue = life.issue;
      }
      const instKey = c.instruction?.path ?? null;
      let loadOrdinal = null;
      if (c.category === 'instruction_loading' && instKey) {
        loadOrdinal = segmentInstructionSeen.has(instKey) ? 'repeat-in-segment' : 'first-in-segment';
        segmentInstructionSeen.add(instKey);
      }
      const event = {
        provider: 'codex', sessionId, timestamp: o.timestamp ?? null, contextSegment: segment,
        issue, attributionMethod: life?.issue ? 'command-target' : (issue ? 'active-binding' : 'unattributed'),
        category: c.category, subcategory: c.subcategory, tool: o.payload.name ?? 'custom_tool',
        command: primary.slice(0, 300), instructionPath: instKey,
        instructionLoadOrdinalInSegment: loadOrdinal, outcome: 'pending-output',
        completeness: 'unavailable', duplicateStatus: 'unchecked', ...loc,
        inputText: command, outputText: '', callId: o.payload.call_id ?? null
      };
      calls.set(o.payload.call_id, event);
      addEvent(event);
    }
    if (o.type === 'response_item' && o.payload?.type === 'custom_tool_call_output') {
      const ev = calls.get(o.payload.call_id);
      if (ev) {
        const out = textOf(o.payload.output);
        ev.outputLine = i + 1;
        ev.outputLocator = { sourceLog: relSource(file), line: i + 1, callId: o.payload.call_id ?? null };
        ev.outputChars = [...out].length;
        ev.outputBytes = bytes(out);
        ev.proxyOutputTokens = Math.ceil(ev.outputChars / 4);
        ev.outputHash = sha256(out);
        ev.contentHash = sha256(`${ev.provider}\n${ev.sessionId}\n${ev.category}\n${ev.inputHash}\n${ev.outputHash}`);
        ev.completeness = outputCompleteness(out);
        ev.outcome = commandOutcome(ev.command, out);
      }
    }
  }
  sessionMeta.set(`codex:${sessionId}`, { issueCount: issues.size, sourceLog: relSource(file) });
}

  let relevant = file.includes('ai-task-manager');

  let activeIssue = null;

  const issues = new Set();
  manifest.push({ provider: 'claude', sourceLog: relSource(file), sessionId, cwd: null, bytes: bytes(text), sha256: sha256(text), lines: lines.length });
  if (!relevant) return;
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    if (!line) continue;
    let o;
    try { o = JSON.parse(line); } catch { continue; }
    sessionId = o.sessionId ?? sessionId;
    cwd = o.cwd ?? cwd;
    const loc = { sourceLog: relSource(file), line: i + 1, ordinal: null, uuid: o.uuid ?? null };
    const timestamp = o.timestamp ?? null;
    if (o.type === 'attachment') {
      const att = o.attachment ?? {};
      const hookText = visibleHookText(att);
      if (/PreCompact|PostCompact|Compact/i.test(`${att.hookEvent ?? ''} ${att.hookName ?? ''}`)) {
        segment += 1;
        activeIssue = null;
        segmentInstructionSeen.clear();
        addEvent({
          provider: 'claude', sessionId, timestamp, contextSegment: segment,
          issue: null, attributionMethod: 'unattributed', category: 'compaction_reset',
          subcategory: 'claude-hook-compact', tool: att.hookName ?? 'hook', command: null,
          instructionPath: null, instructionLoadOrdinalInSegment: null, outcome: 'segment_boundary',
          completeness: 'complete', duplicateStatus: 'unchecked', ...loc, inputText: '', outputText: ''
        });
      }
      if (String(att.type ?? '').startsWith('hook_') || att.hookEvent || att.hookName || ['deferred_tools_delta', 'output_style', 'skill_listing'].includes(att.type)) {

          provider: 'claude', sessionId, timestamp, contextSegment: segment,

          instructionPath: instructionInfo(hookText)?.path ?? null,

          completeness: outputCompleteness(hookText), duplicateStatus: 'unchecked', ...loc,
          inputText: hookText, outputText: ''
        });
      }
      continue;
    }
    if (o.type === 'assistant') {
      for (const item of o.message?.content ?? []) {
        if (item?.type !== 'tool_use') continue;
        const command = item.input?.command ?? item.input?.file_path ?? item.input?.pattern ?? textOf(item.input);
        const parts = splitShell(command);
        const primary = parts.find((p) => classify(p, item.name).category !== 'other_unattributable') ?? parts[0] ?? command;
        const c = classify(primary, item.name);
        const life = c.lifecycle;
        const issue = life?.issue ?? activeIssue;
        if (life?.issue) {
          issues.add(life.issue);
          if (['bind', 'start', 'resume'].includes(life.verb)) activeIssue = life.issue;
        }
        const instKey = c.instruction?.path ?? null;
        let loadOrdinal = null;
        if (c.category === 'instruction_loading' && instKey) {
          loadOrdinal = segmentInstructionSeen.has(instKey) ? 'repeat-in-segment' : 'first-in-segment';
          segmentInstructionSeen.add(instKey);
        }
        const ev = {
          provider: 'claude', sessionId, timestamp, contextSegment: segment,
          issue, attributionMethod: life?.issue ? 'command-target' : (issue ? 'active-binding' : 'unattributed'),
          category: c.category, subcategory: c.subcategory, tool: item.name ?? 'tool',
          command: primary.slice(0, 300), instructionPath: instKey,
          instructionLoadOrdinalInSegment: loadOrdinal, outcome: 'pending-output',
          completeness: 'unavailable', duplicateStatus: 'unchecked', ...loc,
          inputText: command, outputText: '', toolUseId: item.id ?? null
        };
        calls.set(item.id, ev);
        addEvent(ev);
      }
      continue;
    }
    if (o.type === 'user') {
      const content = o.message?.content;
      const arr = Array.isArray(content) ? content : [];
      for (const item of arr) {
        if (item?.type !== 'tool_result') continue;
        const id = item.tool_use_id;
        const ev = calls.get(id);
        if (ev) {
          const out = textOf(item.content ?? o.toolUseResult ?? '');
          ev.outputLine = i + 1;
          ev.outputLocator = { sourceLog: relSource(file), line: i + 1, uuid: o.uuid ?? null, toolUseId: id };
          ev.outputChars = [...out].length;
          ev.outputBytes = bytes(out);
          ev.proxyOutputTokens = Math.ceil(ev.outputChars / 4);
          ev.outputHash = sha256(out);
          ev.contentHash = sha256(`${ev.provider}\n${ev.sessionId}\n${ev.category}\n${ev.inputHash}\n${ev.outputHash}`);
          ev.completeness = item.is_error ? 'complete-error' : outputCompleteness(out);
          ev.outcome = item.is_error ? 'failed_or_refused' : commandOutcome(ev.command, out);
        }
      }
    }
  }
  sessionMeta.set(`claude:${sessionId}`, { issueCount: issues.size, sourceLog: relSource(file) });

function dist(values) {
  const xs = values.filter((x) => Number.isFinite(x)).sort((a, b) => a - b);
  if (!xs.length) return { n: 0, min: null, median: null, p75: null, p90: null, max: null };
  const q = (p) => xs[Math.min(xs.length - 1, Math.floor((xs.length - 1) * p))];
  return { n: xs.length, min: xs[0], median: q(0.5), p75: q(0.75), p90: q(0.9), max: xs[xs.length - 1] };
}

const usable = events.filter((e) => e.duplicateStatus === 'unique-by-visible-content-hash');

  const s = bySession.get(sk);
  s.categories[ev.category] = (s.categories[ev.category] ?? 0) + 1;
  s.chars[ev.category] = (s.chars[ev.category] ?? 0) + ev.inputChars + ev.outputChars;
  s.segments.add(ev.contextSegment);
  if (ev.issue) s.issues.add(ev.issue);
  if (ev.issue && ev.attributionMethod !== 'unattributed') {
    const ik = `${ev.provider}:${ev.issue}`;
    if (!byIssue.has(ik)) byIssue.set(ik, { provider: ev.provider, issue: ev.issue, categories: {}, chars: {}, sessions: new Set(), outcomes: new Set(), segments: new Set() });
    const it = byIssue.get(ik);
    it.categories[ev.category] = (it.categories[ev.category] ?? 0) + 1;
    it.chars[ev.category] = (it.chars[ev.category] ?? 0) + ev.inputChars + ev.outputChars;
    it.sessions.add(ev.sessionId);
    it.segments.add(`${ev.sessionId}:${ev.contextSegment}`);
    it.outcomes.add(ev.outcome);
  }
}

  extractionScript: relSource(path.join(outDir, 'extract-1558-context-burden-v2.mjs')),
  invocation: `node ${path.join(outDir, 'extract-1558-context-burden-v2.mjs')} --cutoff=${EVENT_CUTOFF_ISO}`,
  eventCutoffIso: EVENT_CUTOFF_ISO,

  uniqueEventCount: usable.length,

  sourceLogCount: manifest.length,
  providers: {},
  categoryTotals: {},
  perSessionDistributions: {},
  perIssueDistributions: {},
  closeOutcomes: {},
  contextSegments: {},
  compactionReloads: [],
};
for (const provider of ['codex', 'claude']) {
  const pe = usable.filter((e) => e.provider === provider);
  summary.providers[provider] = {
    sessions: new Set(pe.map((e) => e.sessionId)).size,
    issuesAttributed: new Set(pe.filter((e) => e.issue).map((e) => e.issue)).size,
    contextSegments: new Set(pe.map((e) => `${e.sessionId}:${e.contextSegment}`)).size,
    events: pe.length,
  };
  summary.categoryTotals[provider] = {};
  for (const cat of categories) {
    const ce = pe.filter((e) => e.category === cat);
    summary.categoryTotals[provider][cat] = {
      events: ce.length,
      inputChars: ce.reduce((a, e) => a + e.inputChars, 0),
      outputChars: ce.reduce((a, e) => a + e.outputChars, 0),
      inputBytes: ce.reduce((a, e) => a + e.inputBytes, 0),
      outputBytes: ce.reduce((a, e) => a + e.outputBytes, 0),
      proxyTokens: Math.ceil((ce.reduce((a, e) => a + e.inputChars + e.outputChars, 0)) / 4),
      complete: ce.filter((e) => String(e.completeness).startsWith('complete')).length,
      truncated: ce.filter((e) => e.completeness === 'truncated').length,
      unavailable: ce.filter((e) => e.completeness === 'unavailable').length,
    };
  }
}
for (const cat of categories) {
  summary.perSessionDistributions[cat] = dist([...bySession.values()].map((s) => s.categories[cat] ?? 0));
  summary.perIssueDistributions[cat] = dist([...byIssue.values()].map((s) => s.categories[cat] ?? 0));
}
for (const ev of usable.filter((e) => e.category === 'lifecycle_command' && /close/.test(e.subcategory))) {
  summary.closeOutcomes[ev.outcome] = (summary.closeOutcomes[ev.outcome] ?? 0) + 1;
}
const segmentGroups = new Map();
for (const ev of usable) {
  const key = `${ev.provider}:${ev.sessionId}:${ev.contextSegment}`;
  if (!segmentGroups.has(key)) segmentGroups.set(key, []);
  segmentGroups.get(key).push(ev);
}
for (const [key, evs] of segmentGroups) {
  const ordered = evs.sort((a, b) => (a.line ?? 0) - (b.line ?? 0));
  const firstLife = ordered.find((e) => e.category === 'lifecycle_command');
  const firstLoads = ordered.filter((e) => e.category === 'instruction_loading' && e.instructionLoadOrdinalInSegment === 'first-in-segment' && (!firstLife || e.line < firstLife.line));
  const helpBefore = ordered.filter((e) => e.category === 'help_discovery' && (!firstLife || e.line < firstLife.line));
  if (firstLife || firstLoads.length || helpBefore.length) {
    const [provider, sessionId, segment] = key.split(':');
    summary.compactionReloads.push({
      provider, sessionId, contextSegment: Number(segment),
      firstLifecycleEventId: firstLife?.eventId ?? null,
      firstInstructionLoadsBeforeLifecycle: firstLoads.length,
      helpDiscoveryBeforeLifecycle: helpBefore.length,
      instructionCharsBeforeLifecycle: firstLoads.reduce((a, e) => a + e.inputChars + e.outputChars, 0),
      helpCharsBeforeLifecycle: helpBefore.reduce((a, e) => a + e.inputChars + e.outputChars, 0),
    });
  }
}
summary.compactionReloads = summary.compactionReloads.slice(0, 200);

  .map((x) => ({

  .sort((a, b) => a.totalEvents - b.totalEvents);
const complete = scenarioCandidates.filter((x) => x.workflowObservation === 'confirmed-completed-observed-close');
const medianComplete = complete[Math.floor(complete.length / 2)] ?? scenarioCandidates[Math.floor(scenarioCandidates.length / 2)];
const highHelp = [...complete].sort((a, b) => b.help - a.help)[0] ?? [...scenarioCandidates].sort((a, b) => b.help - a.help)[0];
const compactionPool = scenarioCandidates.filter((x) => x.contextSegments > 1 && x.instructions > 0);
const completeCompactionPool = compactionPool.filter((x) => x.workflowObservation === 'confirmed-completed-observed-close');
const compactionHeavy = [...(completeCompactionPool.length ? completeCompactionPool : compactionPool)].sort((a, b) => b.instructions - a.instructions || b.contextSegments - a.contextSegments || b.help - a.help)[0]
  ?? [...scenarioCandidates].sort((a, b) => b.contextSegments - a.contextSegments || b.instructions - a.instructions)[0];
const heavy = [...scenarioCandidates].sort((a, b) => b.totalEvents - a.totalEvents || b.totalChars - a.totalChars)[0];
const scenarios = {

function tableTotals(provider) {
  return categories.map((cat) => {
    const t = summary.categoryTotals[provider][cat];
    return `| ${cat} | ${t.events} | ${t.inputChars} | ${t.outputChars} | ${t.proxyTokens} | ${t.complete} | ${t.truncated} | ${t.unavailable} |`;
  }).join('\n');
}

## Method

The extractor records one ledger row per visible tool, lifecycle, instruction, hook/host, compaction, or other traffic event. It stores source-log path plus line locator, provider, session ID, timestamp, context segment, issue attribution, category, command/tool, instruction path when identifiable, visible input/output character counts, UTF-8 byte counts, completeness, close outcome, and duplicate status.

Issue attribution is command-target first. If an event has no explicit issue but follows an explicit bind/start/resume in the same observable context segment, it is attributed to that active binding; if the session has multiple issue targets, the method is marked active-binding-shared-session. Otherwise it remains unattributed. Whole-corpus activity is not divided by only the subset of issues with close commands.

Close outcomes are separated as confirmed successful close, failed close, attempted close with unknown output/outcome, and not-close. A confirmed close requires visible output with success language; missing or truncated output is not treated as success.

Deduplication uses exact visible-content hashes across provider, session, category, input, and output. This removes exact repeated/imported log events where visible content matches, but forked or semantically repeated work with different text remains counted and is labeled as remaining uncertainty.

Character counts and bytes are raw visible transcript counts. Proxy tokens use \`Math.ceil(chars / 4)\` and are labeled only as proxy. No real tokenizer was run.

The extraction cutoff is \`${summary.eventCutoffIso}\`, which excludes this audit/extraction session from the measured historical baseline. Re-run with a different \`--cutoff=<ISO>\` only if intentionally changing the evidence window.

## Compaction And Context Segments

Codex compaction is detected from explicit \`type=compacted\` records. Claude compaction detection is limited to visible \`PreCompact\`/\`PostCompact\` hook names; absence of those hook names is undetectable, not evidence that no compaction occurred. Each detected boundary starts a new context segment and clears active issue binding for attribution.

Within each segment, instruction reads are labeled first-in-segment or repeat-in-segment by exact instruction path. The cohort JSON includes the first instruction-load and help/discovery burden before the next lifecycle command for observable segments. These are temporal associations only; they do not prove compaction caused a reload.

Manual validation inspected raw source JSON at the ledger locators for a stratified sample from both providers:

- Codex positive matches: \`ev-0000035\` instruction load, \`ev-0000037\` workflow-guidance search, \`ev-0000043\` lifecycle command, \`ev-0000001\` host instructions, \`ev-0000015\` compaction record.
- Claude positive matches: \`ev-0026126\` instruction load, \`ev-0025658\` workflow-guidance search, \`ev-0026608\` lifecycle command, \`ev-0025541\` hook-injected instruction/guard text, \`ev-0026865\` compact hook boundary.
- Excluded near-matches: Codex \`ev-0000005\` patch text mentioning lifecycle concepts is \`other_unattributable\`; Claude \`ev-0025551\` environment/status probe is \`other_unattributable\`; ordinary implementation searches such as spec reads and spelling/test-code searches remain outside help/discovery unless workflow guidance terms are present.

Validation found and corrected four extraction errors before final output: patch/documentation text was too easily classified as lifecycle/help, \`--help\` and \`aitm help\` rows were being mixed into lifecycle rows instead of help/discovery, shell redirection like \`2>&1\` could create a phantom issue \`#2\`, and duplicate hashes originally ignored returned output. The final ledger includes input line and output line locators where output is visible.

Fail-open checks: unavailable content remains \`completeness=unavailable\` rather than zero burden; missing tool output remains \`outcome=pending-output\` or an attempted-close unknown outcome; duplicate rows are marked \`duplicateStatus=exact-duplicate-visible-content\`; unknown issue attribution remains \`attributionMethod=unattributed\`. In the final all-row ledger, these checks surface 65,496 unavailable rows, 1,562 truncated rows, 157,858 exact duplicates, 229,093 unattributed rows, and 8 command-like rows with no output locator. These are retained as uncertainty, not silently converted to zero.
