% Pandoc's --include-in-header replaces book.json's header-includes instead of
% appending to them, so this selected header must retain the index setup too.
\usepackage{makeidx}
\makeindex
\usepackage{adjustbox}
\usepackage{fancyhdr}
\PassOptionsToPackage{hypertexnames=false}{hyperref}
\setlength{\headheight}{14pt}

\newcommand{\bookchapterheader}{%
  \ifnum\value{chapter}>0 Chapter \thechapter\fi
}

\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\bookchapterheader}
\fancyfoot[R]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyhead[L]{\bookchapterheader}%
  \fancyfoot[R]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

\makeatletter
\newcommand{\booksubtitle}{}
\providecommand{\subtitle}[1]{\gdef\booksubtitle{#1}}

\renewcommand{\maketitle}{%
  \begin{titlepage}
    \thispagestyle{empty}%
    \centering
    \begin{adjustbox}{max width=\textwidth,center}
      \includegraphics{title-page.png}%
    \end{adjustbox}
    \par\vspace{2em}
    {\fontsize{34}{40}\selectfont\bfseries \@title\par}
    \vspace{1em}
    {\Large \booksubtitle\par}
    \vfill
    {\large \@author\par}
  \end{titlepage}%
}
\makeatother

\makeatletter
\renewcommand*\pandocbounded[1]{%
  \sbox\pandoc@box{#1}%
  \Gscale@div\@tempa{.7\textheight}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \makebox[\linewidth][c]{%
    \ifdim\@tempa\p@<\p@
      \scalebox{\@tempa}{\usebox\pandoc@box}%
    \else
      \usebox{\pandoc@box}%
    \fi
  }%
}
\makeatother

\newcommand{\bookchapter}[3]{%
  \clearpage
  \refstepcounter{chapter}%
  \phantomsection
  \addcontentsline{toc}{chapter}{\protect\numberline{\thechapter}#2}%
  \markboth{#2}{#2}%
  \thispagestyle{plain}%
  \noindent
  \begin{adjustbox}{max width=\textwidth,center}
    \includegraphics{#1}%
  \end{adjustbox}
  \par\nobreak\vspace{1.4em}
  \begin{center}
    {\huge\bfseries #2\par}
    \vspace{0.7em}
    {\large\itshape #3\par}
  \end{center}
  \par\vspace{1.4em}\noindent
}
